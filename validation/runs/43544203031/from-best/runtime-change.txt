+class StatementScanner {
+ public:
+  // The cursor and lexical state survive read boundaries. Leave a trailing
+  // byte pending only when a second byte can change its meaning.
+  size_t next_end(const std::string& buf) {
+    for (; cursor_ < buf.size(); ++cursor_) {
+      const char c = buf[cursor_];
+      const bool has_next = cursor_ + 1 < buf.size();
+      const char next = has_next ? buf[cursor_ + 1] : '\0';
+      switch (state_) {
+        case State::LineComment:
+          if (c == '\n') state_ = State::Sql;
+          break;
+        case State::BlockComment:
+          if (c == '*') {
+            if (!has_next) return std::string::npos;
+            if (next == '/') {
+              state_ = State::Sql;
+              ++cursor_;
+            }
+          }
+          break;
+        case State::Quoted:
+          if (c == '\\' && quote_ != '`') {
+            if (!has_next) return std::string::npos;
+            ++cursor_;
+          } else if (c == quote_) {
+            if (!has_next) return std::string::npos;
+            if (next == quote_) ++cursor_;
+            else state_ = State::Sql;
+          }
+          break;
+        case State::Vector:
+          // Match tokenize(): a bare vector consumes balanced brackets;
+          // quotes, comments and semicolons inside it are ordinary data.
+          // Invalid contents are rejected by the parser after the frame ends.
+          if (c == '[') ++vector_depth_;
+          else if (c == ']' && --vector_depth_ == 0) state_ = State::Sql;
+          break;
+        case State::Sql:
+          if ((c == '-' || c == '/') && !has_next) return std::string::npos;
+          if (c == '-' && next == '-') {
+            state_ = State::LineComment;
+            ++cursor_;
+          } else if (c == '#') {
+            state_ = State::LineComment;
+          } else if (c == '/' && next == '*') {
+            state_ = State::BlockComment;
+            ++cursor_;
+          } else if (c == '\'' || c == '"' || c == '`') {
+            state_ = State::Quoted;
+            quote_ = c;
+          } else if (c == '[') {
+            state_ = State::Vector;
+            vector_depth_ = 1;
+          } else if (c == ';') {
+            return cursor_++;
+          }
+          break;
+      }
+    }
+    return std::string::npos;
+  }
+
+  void discard_prefix(size_t count) { cursor_ -= count; }

+ private:
+  enum class State { Sql, Quoted, Vector, LineComment, BlockComment };
+  State state_ = State::Sql;
+  size_t cursor_ = 0;
+  size_t vector_depth_ = 0;
+  char quote_ = '\0';
+};

+// The common execute envelope is a flat object of scalar members. Validate
+// it completely into an owned request, avoiding a DOM and a second scan for id.
+// Anything outside the admitted shape —
+// nesting, backslash escapes, unknown keys, out-of-range integers — falls back
+// to the nlohmann parse below, which stays the reference implementation.
+// Only unescaped ASCII strings use this path. Non-ASCII must go through the
+// reference parser's UTF-8 validation before SQL can have any side effects.
+static bool needs_strict_string_parse(const char* p, const char* q) {
+  while (q - p >= 8) {
+    uint64_t w = 0;
+    std::memcpy(&w, p, 8);
+    if ((w & 0x8080808080808080ull) != 0) return true;
+    if (((w - 0x2020202020202020ull) & ~w & 0x8080808080808080ull) != 0) return true;
+    p += 8;
+  }
+  while (p < q) {
+    const unsigned char c = static_cast<unsigned char>(*p);
+    if (c < 0x20 || c >= 0x80) return true;
+    ++p;
+  }
+  return false;
+}

+static bool scan_simple_json_string(const std::string& t, size_t* p, std::string* out) {
+  size_t i = *p;
+  if (i >= t.size() || t[i] != '"') return false;
+  ++i;
+  // A 250 KB SQL string dominates an ingest request, so the span is found with a
+  // vectorised search and then proven escape-free and control-byte-free instead
+  // of being walked one byte at a time.
+  const char* base = t.data();
+  const char* hit = static_cast<const char*>(std::memchr(base + i, '"', t.size() - i));
+  if (hit == nullptr) return false;
+  const size_t close = static_cast<size_t>(hit - base);
+  if (std::memchr(base + i, '\\', close - i) != nullptr) return false;
+  if (needs_strict_string_parse(base + i, base + close)) return false;
+  if (out) out->assign(base + i, close - i);
+  *p = close + 1;
+  return true;
+}

+static Json strict_json(const std::string& text, bool try_fast = true) {
+  if (try_fast) {
+    Json fast;
+    if (fast_envelope(text, &fast)) return fast;
+  }
+  std::vector<std::set<std::string>> keys;
+  Json parsed=Json::parse(text, [&](int, Json::parse_event_t event, Json& value) {
+    if (event == Json::parse_event_t::object_start) keys.emplace_back();
+    else if (event == Json::parse_event_t::object_end) keys.pop_back();
+    else if (event == Json::parse_event_t::key &&
+             !keys.back().insert(value.get<std::string>()).second)
+      throw std::runtime_error("duplicate JSON object key");
+    return true;

+  // The transport is length-delimited: never execute a valid prefix followed
+  // by an unvalidated NUL/tail. Keep the parser's existing rejection priority;
+  // escaped \u0000 contains no raw NUL and remains an ordinary JSON string.

+// Called only after strict_json validated the entire request. Retain the raw
+// top-level id value, including integers wider than uint64, instead of
+// round-tripping through the JSON library's uint64/double representation.
+static std::string raw_request_id(const std::string& text) {
+  auto skip_space=[&](size_t& p) {
+    while (p<text.size() && std::isspace(static_cast<unsigned char>(text[p]))) ++p;
+  };
+  auto string_end=[&](size_t p) {
+    ++p;
+    while (p<text.size()) {
+      if (text[p]=='\\') p+=2;
+      else if (text[p++]=='"') return p;
+    }
+    throw std::runtime_error("unterminated JSON string");
+  };
+  size_t p=0; skip_space(p); ++p;
+  for (;;) {
+    skip_space(p);
+    if (p>=text.size() || text[p]=='}') return "null";
+    const size_t key_start=p;
+    p=string_end(p);
+    const std::string key=Json::parse(text.substr(key_start,p-key_start)).get<std::string>();
+    skip_space(p); ++p; skip_space(p);
+    const size_t value_start=p;
+    size_t depth=0;
+    while (p<text.size()) {
+      const char c=text[p];
+      if (c=='"') { p=string_end(p); continue; }
+      if (c=='[' || c=='{') ++depth;
+      else if (c==']' || c=='}') {
+        if (!depth) break;
+        --depth;
+      } else if (c==',' && !depth) break;
+      ++p;
+    }
+    if (key=="id") return text.substr(value_start,p-value_start);
+    if (p>=text.size() || text[p]=='}') return "null";
+    ++p;
+  }

+enum class Reason { accepted, environment, syntax_or_size, midpoint };
+struct Result {
+  Reason reason;
+  const char* ptr;
+  float value;
+  bool accepted() const { return reason==Reason::accepted; }

+// Construct once per vector. The caller must not change the FP environment
+// during that vector's parse. Unsupported builds/environments use old parsing.
+class Context {
+ public:
+  explicit Context(bool attempt=true): enabled_(attempt && supported()) {}
+  bool enabled() const { return enabled_; }
+  static bool supported() {
+#if defined(__x86_64__) && defined(__SSE2__) && FLT_EVAL_METHOD == 0 && !defined(__FAST_MATH__)
+    if(!std::numeric_limits<float>::is_iec559 ||
+       !std::numeric_limits<double>::is_iec559 || sizeof(float)!=4 || sizeof(double)!=8)
+      return false;
+    if(std::fegetround()!=FE_TONEAREST)return false;
+    const unsigned csr=_mm_getcsr();
+    return (csr & ((3u<<13)|(1u<<15)|(1u<<6)))==0 &&
+           (csr & (63u<<7))==(63u<<7);
+#else
+    return false;
+#endif
+  }
+  Result parse(const char* begin,const char* end) const {
+    if(!enabled_)return {Reason::environment,begin,0};
+    const char* p=begin;
+    bool negative=false;
+    if(p!=end && *p=='-'){negative=true;++p;}
+    // A deliberately strict subset: at least one integer digit, and at least
+    // one fractional digit when a dot is present. .1/1./+1 all fall back.
+    if(p==end || *p<'0' || *p>'9')return {Reason::syntax_or_size,begin,0};
+    uint64_t mantissa=0;
+    unsigned digits=0,fraction=0;
+    while(p!=end && *p>='0' && *p<='9') {
+      if(digits==15)return {Reason::syntax_or_size,begin,0};
+      mantissa=mantissa*10+static_cast<unsigned>(*p-'0');++digits;++p;
+    }

+      while(p!=end && *p>='0' && *p<='9') {
